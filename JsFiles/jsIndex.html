Number Guessing Game


const minNum = 1;
const maxNum = 100;
const answer = Math.floor(Math.random() * (maxNum - minNum + 1));
//Si temonos un numero minimo que no sea 1, debemos añadir un + minNUm al Math.foor()


let attemps = 0;
let guess;
let running = true; 

while(running){

    guess = window.prompt(`Guess a number entre ${minNum} - ${maxNum}: `)

    if(guess === null) {
        window.alert("Juego cancelado. ¡Hasta la próxima!");
        running = false;
        continue; // Salta al siguiente ciclo (que no ocurrirá porque running es false)
    }


    guess = Number(guess);
    
    if(isNaN(guess)){
        window.alert("Pon un número válido!")
    }
    else if (guess < minNum || guess > maxNum){
        window.alert("No te salgas del rango!");
    }
    else{
        attemps++;
        if(guess < answer){
                window.alert("Too low");
        }
        else if(guess > answer){
                window.alert("To high")
        }
        else{
            window.alert(`CORRECT! the answer was ${answer}. it took you ${attemps} attemps.`);
            running = false;
        }
    }  
}

Random password generator


        // Función 
function generatePassword(length, includeLowercase, includeUppercase, includeNumbers, includeSymbols) {

    // Definimos los caracteres disponibles para cada categoría
    const lowercaseChars = "abcdefghijklmnñopqrstuvwxyz";  // Letras minúsculas
    const upercaseChars = "ABCDEFGHIJKLMNÑOPQRSTUVWXYZ";   // Letras mayúsculas 
    const numberChars = "0123456789";                      // Números
    const symbolChars = "!#~$%&/()={}'`¡@º][*";           // Símbolos especiales

    let allowedChars = "";  // Aquí se almacenarán todos los caracteres permitidos
    let password = "";      // Variable donde se construirá la contraseña final

    // Concatenamos los caracteres permitidos según los parámetros:
    allowedChars += includeLowercase ? lowercaseChars : "";  // Si includeLowercase es true, añade minúsculas
    allowedChars += includeUppercase ? upercaseChars : "";   // Si includeUppercase es true, añade mayúsculas
    allowedChars += includeNumbers ? numberChars : "";       // Si includeNumbers es true, añade números
    allowedChars += includeSymbols ? symbolChars : "";      // Si includeSymbols es true, añade símbolos

    // Validaciones:
    if (length <= 0) {
        return "(Password length must be at least 1)";  // La longitud debe ser mayor a 0
    }
    if (allowedChars.length === 0) {
        return "(At least 1 set of character needs to be selected)";  // Debe haber al menos un tipo de carácter
    }

    // Generamos la contraseña carácter por carácter:
    for (let index = 0; index < length; index++) {
        const randomIndex = Math.floor(Math.random() * allowedChars.length);  // Índice aleatorio
        password += allowedChars[randomIndex];  // Añadimos un carácter aleatorio de allowedChars
    }

    return password;  // Retornamos la contraseña generada
}

***************
// Parámetros de ejemplo para generar una contraseña:
const passwordLength = 12;
const includeLowercase = true;
const includeUppercase = true;
const includeNumbers = true;
const includeSymbols = true;

// Llamada a la función:
const password = generatePassword(
    passwordLength,
    includeLowercase,
    includeUppercase,
    includeNumbers,
    includeSymbols
);

// Mostramos la contraseña generada:
console.log(`Generated password: ${password}`);
