Math object

Para hacer operaciones matemáticas como Pi, elevados, max, min, ángulos, raices, abs..

let a = 3.21; let b = 2; let c;

  • c = Math.round(a);
  • c = Math.floor(a);
  • c = Math.ceil(a);
  • c = Math.min(a, b, c);
  • c = Math.max(a, b, c);
  • c = Math.trunc(a);
  • c = Math.pow(3,4); //3 elevado a 4
  • c =Math.sqrt(81);
  • c =Math.abs(a); //el positivo
